function doGet(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (e && e.parameter && e.parameter.action === "ping") {
      return ContentService.createTextOutput(JSON.stringify({success: true, message: "pong"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var dataString = e.parameter.data;
    if (!dataString) {
      return ContentService.createTextOutput(JSON.stringify({success: false, error: "No data provided"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = JSON.parse(dataString);
    var action = data.action;
    
    // ==========================================
    // 1. บันทึกประวัติการใช้งาน (logHistory)
    // ==========================================
    if (action === "logHistory") {
      var sheet = ss.getSheetByName("History") || ss.getSheetByName("HistoryLog") || ss.getSheets()[0];
      var timestamp = new Date();
      sheet.appendRow([timestamp, data.memberID, data.name, data.keyID, data.event]);
      return ContentService.createTextOutput(JSON.stringify({success: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // ==========================================
    // 2. บันทึกเหตุการณ์ยืม-คืน (borrowReturn)
    // ==========================================
    else if (action === "borrowReturn") {
      var sheet = ss.getSheetByName("History") || ss.getSheetByName("HistoryLog") || ss.getSheets()[0];
      var timestamp = new Date();
      sheet.appendRow([timestamp, data.memberID, data.name, data.keyID, data.event]);
      return ContentService.createTextOutput(JSON.stringify({success: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // ==========================================
    // 3. อัปเดตสถานะกุญแจ (updateKeyStatus)
    // ==========================================
    else if (action === "updateKeyStatus") {
      var sheet = ss.getSheetByName("Keys") || ss.getSheetByName("KeyStatus");
      if (sheet) {
        var rows = sheet.getDataRange().getValues();
        for (var i = 1; i < rows.length; i++) {
          if (rows[i][0] == data.keyID) {
            sheet.getRange(i + 1, 2).setValue(data.status);
            sheet.getRange(i + 1, 3).setValue(data.borrower);
            break;
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // ==========================================
    // 4. เพิ่มสมาชิกใหม่ (addMember) - [แก้ไขให้ตรงกับหัวตารางเป๊ะๆ]
    // ==========================================
    else if (action === "addMember") {
      var sheet = ss.getSheetByName("Members");
      if (!sheet) {
        sheet = ss.insertSheet("Members");
        sheet.appendRow(["MemberID", "Name", "FingerID", "Role", "Status", "Phone", "Timestamp"]);
      }
      var timestamp = new Date();
      
      // เรียงลำดับตามหัวตารางของคุณ: MemberID, Name, FingerID, Role, Status, Phone, Timestamp
      sheet.appendRow([
        data.memberID || "",
        data.name || "",
        data.fingerID || "",
        data.role || "Member",
        data.status || "Active",
        data.phone || "",
        timestamp
      ]);
      
      return ContentService.createTextOutput(JSON.stringify({success: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // ==========================================
    // 5. ลบสมาชิก (deleteMember)
    // ==========================================
    else if (action === "deleteMember") {
      var sheet = ss.getSheetByName("Members");
      if (sheet) {
        var rows = sheet.getDataRange().getValues();
        for (var i = 1; i < rows.length; i++) {
          if (rows[i][0] == data.memberID) {
            sheet.deleteRow(i + 1);
            break;
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({success: false, error: "Invalid action"}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({success: false, error: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
